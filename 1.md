frontend:
    1. git clone --depth=1 https://github.com/pure-admin/pure-admin-thin.git .
    2. pnpm i
    3. pnpm run dev

backend:
    1. 进入目录并克隆代码：
       cd /Volumes/ORICO/Desktop_backup/test/mini-agent/backend
       git clone --depth=1 https://github.com/Massad/gin-boilerplate.git .

    2. 配置国内 Go 代理（解决国内拉取依赖与 Toolchain 出现 TLS handshake timeout 超时问题）：
       go env -w GOPROXY=https://goproxy.cn,direct

    3. 下载并整理依赖包：
       go mod tidy

    4. 创建并配置环境变量：
       cp .env_rename_me .env

       配置说明（修改 .env）：
       - 解决 "ACCESS_SECRET and REFRESH_SECRET must be different" 报错：
         运行两次 `openssl rand -hex 32` 生成两个不同的密钥，分别填入 ACCESS_SECRET 与 REFRESH_SECRET。
       - 解决 "pq: SSL is not enabled on the server" 报错：
         本地 PostgreSQL 默认未开启 SSL，需修改为：
         SSL=FALSE

    5. 启动依赖服务（PostgreSQL & Redis）与初始化数据库：
       # 启动服务
       brew services start postgresql@14
       brew services start redis

       # 创建数据库并导入初始表结构
       createdb golang_gin_db
       psql -d golang_gin_db -f db/database.sql

    6. 运行后端服务：
       go run main.go

       - 服务端口：http://localhost:9000
       - Swagger 文档：http://localhost:9000/swagger/index.html
